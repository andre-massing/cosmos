# Get NGSolve

mkdir src
cd src
git clone https://github.com/ngsolve/ngsolve
git checkout 7c07f12
cd ngsolve
git submodule update --init --recursive

cd ../..
mkdir build inst
cd build

# BUILD NGSOLVE using cmake (details may depend on your machine and os / linux distribution)
# For further cmake options and instructions, see the NGSolve documentation at
# https://ngsolve.org/docu/v6.2.1906/install/install_sources.html

cmake \
-DCMAKE_BUILD_TYPE=RELEASE \
-DCMAKE_INSTALL_PREFIX=../inst \
-DUSE_GUI=ON \
-DUSE_UMFPACK=ON \
-DMKL_ROOT=${MKLROOT} \
-DUSE_MKL=ON \
-DCMAKE_CXX_COMPILER=g++ \
-DCMAKE_C_COMPILER=gcc \
../src/ngsolve

make 
make install

# Now the executables in inst/bin have to be added to the PATH 
# and the location of the python libraries have to be added to the PYTHONPATH
